cmake_minimum_required(VERSION 3.16)
project(NoeMixerrMCP LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt5 QUIET COMPONENTS Core Xml)
if(Qt5_FOUND)
  set(NOEAI_QT Qt5)
else()
  find_package(Qt6 REQUIRED COMPONENTS Core Xml)
  set(NOEAI_QT Qt6)
endif()
add_executable(noemixerr-mcp main.cpp)
target_link_libraries(noemixerr-mcp PRIVATE ${NOEAI_QT}::Core ${NOEAI_QT}::Xml)
if(MSVC)
  target_compile_options(noemixerr-mcp PRIVATE /W4)
else()
  target_compile_options(noemixerr-mcp PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME MCPHello COMMAND noemixerr-mcp --version)
