cmake_minimum_required(VERSION 3.20)
project(NoeMixerrRecorder VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PORTAUDIO REQUIRED IMPORTED_TARGET portaudio-2.0)
find_package(Threads REQUIRED)
add_executable(noemixerr-recorder main.cpp)
target_link_libraries(noemixerr-recorder PRIVATE PkgConfig::PORTAUDIO Threads::Threads)
if(MSVC)
  target_compile_options(noemixerr-recorder PRIVATE /W4)
else()
  target_compile_options(noemixerr-recorder PRIVATE -Wall -Wextra -Wpedantic)
endif()
