cmake_minimum_required(VERSION 3.20)
project(NoeMixerr VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(noemixerr_core STATIC src/core.cpp)
target_include_directories(noemixerr_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
 target_compile_options(noemixerr_core PRIVATE /W4)
else()
 target_compile_options(noemixerr_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(NOEMIXERR_GUI "Build Qt6 desktop DAW" ON)
if(NOEMIXERR_GUI)
 find_package(Qt6 QUIET COMPONENTS Widgets Multimedia)
 if(Qt6_FOUND)
  add_executable(NoeMixerr src/gui.cpp)
  target_link_libraries(NoeMixerr PRIVATE noemixerr_core Qt6::Widgets Qt6::Multimedia)
  set_target_properties(NoeMixerr PROPERTIES WIN32_EXECUTABLE TRUE MACOSX_BUNDLE TRUE)
 else()
  message(STATUS "Qt6 Widgets/Multimedia unavailable, building core and tests")
 endif()
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(noemixerr_tests tests/core_tests.cpp)
 target_link_libraries(noemixerr_tests PRIVATE noemixerr_core)
 add_test(NAME core COMMAND noemixerr_tests)
endif()
